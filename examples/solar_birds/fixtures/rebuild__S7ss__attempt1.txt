```python
import json, os, warnings
import pandas as pd
import pyfixest as pf
warnings.filterwarnings("ignore")
df = pd.read_parquet(os.path.join(os.environ["WORK_DIR"], "analysis.parquet"))
d = df
# ivreghdfe ShannonBD (PI = Sun_ccpu) controls, absorb(county year) cluster(county) keepsingleton
m = pf.feols("ShannonBD ~ Temp + Wind + Pop + Duration + Carbon + Water + Green + Farm + Grass | county + ym | PI ~ Sun_ccpu", d, vcov={"CRV1": "county"}, fixef_rm="none")
print("RESULT_JSON: " + json.dumps({"b": float(m.coef()["PI"]), "se": float(m.se()["PI"]), "n": int(m._N)}))
```
